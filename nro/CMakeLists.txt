cmake_minimum_required(VERSION 3.13)
project(autochart-hero LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(autochart-hero source/main.cpp third_party/qrcodegen.cpp)
set(PORTLIBS "$ENV{DEVKITPRO}/portlibs/switch")
target_include_directories(autochart-hero PRIVATE third_party "${PORTLIBS}/include")
target_link_directories(autochart-hero PRIVATE "${PORTLIBS}/lib")
target_link_libraries(autochart-hero PRIVATE curl mbedtls mbedx509 mbedcrypto z nx)
nx_generate_nacp(autochart-hero.nacp NAME "Autochart Hero" AUTHOR "Jul879n" VERSION "2.0.7")
nx_create_nro(autochart-hero NACP autochart-hero.nacp ICON "${CMAKE_CURRENT_SOURCE_DIR}/icon.jpg" ROMFS "${CMAKE_CURRENT_SOURCE_DIR}/romfs")
