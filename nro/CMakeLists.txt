cmake_minimum_required(VERSION 3.13)
project(sh-installer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(sh-installer source/main.cpp third_party/qrcodegen.cpp)
set(PORTLIBS "$ENV{DEVKITPRO}/portlibs/switch")
target_include_directories(sh-installer PRIVATE third_party "${PORTLIBS}/include")
target_link_directories(sh-installer PRIVATE "${PORTLIBS}/lib")
target_link_libraries(sh-installer PRIVATE curl mbedtls mbedx509 mbedcrypto z nx)
nx_generate_nacp(sh-installer.nacp NAME "SH Installer" AUTHOR "Switch Hero Autochart" VERSION "2.0.2")
nx_create_nro(sh-installer NACP sh-installer.nacp ROMFS "${CMAKE_CURRENT_SOURCE_DIR}/romfs")
